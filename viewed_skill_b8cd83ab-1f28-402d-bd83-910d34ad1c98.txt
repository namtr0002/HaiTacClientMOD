}

GameCanvas.ClearkeyMove(3);
}
if (selectedHieuUngIndex < 0) selectedHieuUngIndex = 0;
if (selectedHieuUngIndex >= totalEff) selectedHieuUngIndex = totalEff - 1;
selectedHieuUngInfo = (HieuUngInfo)HieuUngManager.vecHieuUng.elementAt(selectedHieuUngIndex);

if (hieuUngList != null) {

if (itemY + itemH > hieuUngList.AC + hieuUngList.maxH) hieuUngList.AB = itemY + itemH - hieuUngList.maxH;
}
updateSoftKeys();
}
}
